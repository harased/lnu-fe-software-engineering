sudo apt update - оновлення ос
sudo apt install example - встановлення пакета по назву
fastfetch - для збору інфи про систему
ls - показує список файлів і папок у поточній директорії
mkdir - створює нову директорію
cd - перейти в директорію
touch file.ex - створює файл
nano file.ex - відкриває файл в текстовому редакторі
echo "text" > file.ex - перенаправляє текст з терміналу прямо в файл та повністю перезаписує його
echo "text" >> file.ex - перенаправляє текст з терміналу прямо в файл додає в кінець строчки новий текст
cat file - виводить текст з файлу
grep "file" directory - для пошуку тексту, у файлі або директорії
ls | grep ".img" - шукає в поточній директорії усі файли де містить .img
history - список усіх команд
ctrl + u - видаляє увесь рядок
ctrl + w - удаляє одне слово
ctrl + l - чистить увесь термінал
rm file.ex - видаляє файл
rm -rf directory - видаляє директорію з усім вмістом
ls -l - виводить усю інформацію про файлів
перший символ є типом файла - це звичайний файл, d - це директорія
далі йдуть дії які є дозволені до цього файлу - rwx
read - читати
write - писати
execute - виконувати
-rwxr-xr--
перші 3 символи (rwx) - права власника файлу
другі 3 символи (r-x)  - права групи власника 
треті 3 символи (r--) - права для усіх решти 
chmod 777 file - видає всі права для усіх
chmod 000 file - забирає всі права для усіх
r = 4
w = 2
x = 1
наприклад якщо я хочу видати усі права я сумую 4+2+1=7 (rwx), якщо я хочу щоб можна було тільки читати та виконувати 4+1=5 (r-x)
./file.ex - виконання файлу у поточній директорії
./projects/app/file.ex - виконання файлу використання директорії, та файлу
cp file.ex file_copy.ex - копіює файл у поточній директорії
cp file.ex ./projects - копіює файл у вказану директорію
mv file.ex file_new_name.ex - перейменує файл
mv file.ex ./projects - перемістить файл у вказану директорію
find /projects -name "filename" - шукає файл за назвою у поточній директорії та у всіх її каталогах
sudo poweroff - виключає машину
